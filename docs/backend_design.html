1. 개요

VYBE v1 백엔드 설계 문서는 사용자 앱(Flutter · 99개 화면 · 상태 변형 187)과 관리자 페이지(업주용 SPA · 32개 화면 · 상태 변형 176 · 조회 섹션 14) 전체를 Firebase 기반 백엔드로 옮긴 결과다. 베타 설계 문서(firebase_structure.html · 2026-06-26)의 구조 · 표기 · 이름 규칙을 그대로 따르고, 베타에서 재사용하는 부분에는 표시를 붙이지 않는다.

2. 전체 아키텍처

클라이언트는 Firestore 구독(읽기)과 onCall 함수(쓰기·확정)만 사용하고, 금액·결제·상태·순번·시간 조건을 바꾸는 모든 경로는 Cloud Functions 안에 있다. 외부 서비스는 점선 상자(모두 [임시])이며 Firebase와 연결되는 지점은 함수(웹훅·호출)뿐이다.

9. 실시간 처리

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ───────── 공통 헬퍼 ─────────
    function signedIn() { return request.auth != null; }
    function uid() { return request.auth.uid; }
    function isAdmin() { return signedIn() && request.auth.token.admin == true; }                          // VYBE 운영 어드민 (베타)
    function isPartner() { return signedIn() && request.auth.token.partner == true; }
    function partnerOf(clubId) { return isPartner() && clubId in request.auth.token.clubIds; }              // 소속 클럽
    function roleIs(roles) { return isPartner() && request.auth.token.role in roles; }
    function managerOf(clubId) { return partnerOf(clubId) && roleIs(['owner','manager']); }
    function ownerOf(clubId) { return partnerOf(clubId) && roleIs(['owner']); }
    function onlyKeys(keys) { return request.resource.data.diff(resource.data).affectedKeys().hasOnly(keys); }
    function isActiveUser() { return signedIn() && exists(/databases/$(database)/documents/users/$(uid()))
                              && get(/databases/$(database)/documents/users/$(uid())).data.status == 'active'; }   // 1 read · 캐시

    // ───────── users (베타 + 서버 전용 키 추가) ─────────
    match /users/{userId} {
      allow read: if signedIn() && (uid() == userId || isAdmin());
      allow create: if false;                                               // 함수(onUserCreated)만
      allow update: if signedIn() && uid() == userId
        && !onlyKeys(['uid','phoneNumber','provider','isIdentityVerified','status','deletionScheduledAt','createdAt',
                      'activeCounts','unreadNotificationCount','shareLockUntil']).hasAny(request.resource.data.diff(resource.data).affectedKeys())
        && onlyKeys(['nickname','profileImageUrl','gender','birthDate','agreements','notificationSettings','lastSeenAt','updatedAt']);
      allow delete: if false;

      match /searchHistory/{historyId} { allow read, write: if signedIn() && uid() == userId; }            // 베타
      match /devices/{deviceId}     { allow read, write: if signedIn() && uid() == userId; }
      match /notifications/{nid}    { allow read: if signedIn() && uid() == userId;
                                      allow update: if signedIn() && uid() == userId && onlyKeys(['read']); // 읽음만
                                      allow create, delete: if false; }
      match /freeEntryAlerts/{clubId} { allow read, write: if signedIn() && uid() == userId
                                        && request.resource.data.keys().hasOnly(['clubId','enabled','updatedAt']); }
      match /history/{itemId}       { allow read: if signedIn() && uid() == userId;
                                      allow update: if signedIn() && uid() == userId && onlyKeys(['hiddenByUser']);
                                      allow create, delete: if false; }
    }

    // ───────── 베타 top-level (그대로) ─────────
    match /nicknames/{key} { allow read, write: if false; }
    match /favorites/{favoriteId} {
      allow read: if signedIn() && (resource.data.userId == uid() || isAdmin());
      allow create: if signedIn() && request.resource.data.userId == uid()
        && request.resource.data.keys().hasOnly(['userId','clubId','createdAt']);
      allow delete: if signedIn() && resource.data.userId == uid();
      allow update: if false;
